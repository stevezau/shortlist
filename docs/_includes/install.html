docker run -d --name shortlist -p 5959:5959 \
  -e TZ=Etc/UTC -e PUID=1000 -e PGID=1000 \
  -v /path/to/shortlist/config:/config \
  --restart unless-stopped \
  {{ site.ghcr }}:latest
{%- unless include.bare %}

Change /path/to/shortlist/config to a folder on the host, then open http://your-host:5959.

{%- endunless -%} {%- else -%}
docker-compose.yml
services:
  shortlist:
    image: {{ site.ghcr }}:latest
    ports: ["5959:5959"]
    volumes: ["./config:/config"]
    environment:
      - TZ=Etc/UTC
      - PUID=1000
      - PGID=1000
    restart: unless-stopped
{%- unless include.bare %}

Save it as docker-compose.yml, run docker compose up -d, then open http://your-host:5959.
